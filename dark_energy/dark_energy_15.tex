\begin{frame}
  \frametitle{The evolution of the universe}
  \framesubtitle{Two possibilities}

  \begin{tikzpicture}[
      declare function={%
        arcsinh(\x) = ln(\x + sqrt(\x^2+1.0));
        arcsin(\x) = \pii/180.0 * asin(\x);
        open0(\x,\om,\ok) = (sqrt(\x*\om*(\x*\ok+\om))-\om*sqrt(\om/\ok)*arcsinh(sqrt(\x*\ok/\om)))/\ok/sqrt(\om);
        open(\x,\om,\ok) = open0(\x,\om,\ok) - open0(1.0,\om,\ok);
        closed0(\x,\om,\ok) = (\ok*sqrt(\x*(\x*\ok + \om)) + sqrt(-\ok)*\omc*arcsin(sqrt(-\x*\ok/\om)))/\ok^2;
        closed(\x,\om,\ok) = closed0(\x,\om,\ok)-closed0(1.0,\om,\ok);
        tclose(\om,\ok) = \pii*\om/(-\ok)^1.5;
        maxa(\om,\ok) = -\om/\ok;
        mint(\omc,\okc,\omo,\oko) = min(closed(0.0,\omc,\okc),open(0.0,\omo,\oko));
        maxt(\om,\ok) = tclose(\omc,\okc) - closed0(1.0,\om,\ok);
      },
    ]
    \begin{axis}[
        ticks=none,
