\begin{frame}
	\frametitle{The Big Bang}
	\framesubtitle{Galactic Redshift}

	It appears that:

	\begin{enumerate}
		\item All Galaxies are moving away from us
		\item The Further away they are, the faster they are moving away
		\item If we were to go to another galaxy, we would see exactly the same thing
	\end{enumerate}

\end{frame}

\begin{frame}
	\frametitle{Constructing a theory of the Universe}
	\framesubtitle{Extended Copernican Principle}
	
	Philosophically, we expect the universe to be:

	\begin{description}
		\item[Homogeneous:]{~ No point in the universe is special.}
		\item[Isotropic:]{~ ~ \  No direction in space is special.}
	\end{description}

	\vspace{1cm}

	If we demand these condititions we find that there are three types of universe:
	\begin{itemize}
		\item Open
		\item Flat
		\item Closed
	\end{itemize}
	
\end{frame}

\begin{frame}
	\frametitle{Dark Energy}
	\framesubtitle{The Problem}
	\begin{itemize}
		\item The Universe is not slowing down in it's expansion, but {\em speeding up!}
		\item One of the ways in which this could happen would be the presence of an energy permeating the universe,  accelerating the expansion.
		\item We call this Dark Energy
		\item Like Dark Matter, it's really just a name for a phenomenon which we observe, and haven't been able to explain yet
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Why is Astronomy important?}
	\framesubtitle{Why put money into research which is explicitly useless?}
	\begin{enumerate}
		\item The techniques which we create in order to study these systems are {\em very} useful. \label{enum:techniques}
		\item If there is a theoretical advance in Quantum Gravity, the experimental proof will come from Cosmology. \label{enum:qg}
		\item Discovery is important. \label{enum:discovery}
	\end{enumerate}
\end{frame}

\begin{frame}
	\frametitle{Why is Astronomy important?}
	\framesubtitle{\ref{enum:techniques}.  Theoretical \& Computational Techniques}
	\begin{itemize}
		\item Massive Projects generate novel ideas in computing, instrumentation and theory.
		\item e.g:\ The Square Kilometer Array
			\begin{itemize}
				\item Radio telescope to be built in South Africa and New Zealand.
				\item Will generate PBs of data per second $\sim100$ times global internet traffic.
				%\item We are building the world's biggest supercomputer to deal with this\ldo 
			\end{itemize}
	\end{itemize}
	\includegraphics[width=\textwidth]{Images/ska.jpg}
\end{frame}

\begin{frame}
	
	\frametitle{Why is Astronomy important?}
	\framesubtitle{\ref{enum:techniques}.  Theoretical \& Computational Techniques}
	Historically, Astronomy has provided important stepping stones for human progress;
	\begin{itemize}
		\item Calendars \& timekeeping
		\item Much of mathematics
			\begin{itemize}
				\item Hipparchus $\rightarrow$ Trigonometry
				\item Newton $\rightarrow$ Calculus
				\item Logarithms
			\end{itemize}
		\item Navigational techniques
		\item Driving development of computers
		\item Space Age
		\item Radio astronomy $\rightarrow$ communications
		\item Optical Astronomy $\rightarrow$ medical imaging
		\item \ldots
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Why is Astronomy important?}
	\framesubtitle{\ref{enum:qg}.  Quantum Gravity}

	\begin{column}{0.6\textwidth}
	\begin{itemize}
		\item Quantum Mechanics and General Relativity are fundamentally incompatible.
		\item To create a unified theory, we should study a system which needs both.
		\item The big bang provides an ideal particle accelerator.
		\item Cosmology is the perfect playground to test out \& discover quantum theories of gravity.
	\end{itemize}
	\end{column}
	\begin{column}{0.4\textwidth}
	\includegraphics[width=\textwidth]{Images/calabiyau.png}
	\end{column}
\end{frame}
