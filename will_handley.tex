\begin{frame}
    \frametitle{Science on the JWST}
    As a space-based observatory it aims to
    \begin{itemize}
        \item to search for light from the first stars and galaxies that formed in the universe after the Big Bang
        \item to study galaxy formation and evolution
        \item to understand star formation and planet formation
        \item to study planetary systems and the origins of life
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Scientific instruments}
    \begin{itemize}
        \item 17 different ``modes'' spread across 4 instruments
    \end{itemize}
    \begin{description}
        \item[NIRCam] Near Infrared Camera (10$\times$4MegaPixels HgCdTe)
            \begin{itemize}
                \item Most of front-page images are with this in its camera mode
                \item Also a spectrometer \& coronograph
                \item 29 filters \& resolutions
                \item Cooled to 37K by sunshield
            \end{itemize}
        \item[NIRSpec] Near Infrared Spectrograph. (2$\times$4MegaPixels). \\
            \begin{itemize}
                \item Three resolution modes in grating wheel assembly + 5 filters
            \end{itemize}
        \item[MIRI] Mid-Infrared instrument
            \begin{itemize}
                \item Cooled to $<$7K by cryocooler
                \item Builds on legacy of Hubble + coronography
            \end{itemize}
        \item[NIRISS] Near Infrared Imager and Slitless Spectrograph
            \begin{itemize}
                \item Single object spectroscopy
                \item Aperture masking interferometry
                \item Backup to NIRCam
            \end{itemize}
    \end{description}
\end{frame}
